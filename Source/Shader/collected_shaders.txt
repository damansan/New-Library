Texture2D baseMap              : register(t0); // Albedo
Texture2D normalMap            : register(t1); // –@üƒ}ƒbƒv
Texture2D metallicRoughnessMap : register(t2); // G:Roughness B:Metallic
Texture2D occlusionMap         : register(t3); // R:Occlusion
Texture2D emissiveMap          : register(t4); // Emissive

SamplerState samplerState      : register(s0);

cbuffer CbObject : register(b2)
{
    row_major float4x4 world;
    float alphaCutoff;
    float3 padding;
    float4 baseColorFactor; // ƒ}ƒeƒŠƒAƒ‹‚ÌŠî–{FiƒŠƒjƒA‹óŠÔj
};

struct VS_OUT
{
    float4 position : SV_POSITION;
    float3 normal   : NORMAL;
    float4 tangent  : TANGENT;
    float2 texcoord : TEXCOORD;
    float3 worldPos : WORLD_POSITION;
};

// o—Í\‘¢‘Ì
struct PS_OUT
{
    float4 albedo            : SV_TARGET0; // Albedo
    float4 normal            : SV_TARGET1; // Normaliƒ[ƒ‹ƒh‹óŠÔj
    float4 position          : SV_TARGET2; // WorldPosition
    float4 roughnessMetallic : SV_TARGET3; // R:Roughness G:Metallic
    float4 emissive          : SV_TARGET4; // Emissive
    float occlusion          : SV_TARGET5; // Occlusion
};

PS_OUT main(VS_OUT pin)
{
    PS_OUT pout;

    // Albedo
    float4 baseColor = baseMap.Sample(samplerState, pin.texcoord);

    // ƒAƒ‹ƒtƒ@‚ÍŒW”‚ğŠ|‚¯‚Ä‚©‚çƒNƒŠƒbƒv”»’è‚·‚é
    baseColor.a *= baseColorFactor.a;

    // MaskƒNƒŠƒbƒv
    clip(baseColor.a - alphaCutoff);

    // ƒeƒNƒXƒ`ƒƒ‚ÍsRGB‚È‚Ì‚ÅƒŠƒjƒA‚É–ß‚µ‚Ä‚©‚çAƒŠƒjƒA‚ÌŒW”‚ğŠ|‚¯‚é
    baseColor.rgb = pow(max(baseColor.rgb, 0.0001f), 2.2f);
    baseColor.rgb *= baseColorFactor.rgb;
    pout.albedo = baseColor;
    
    // –@üƒ}ƒbƒsƒ“ƒO
    float3 N = normalize(pin.normal);
    float3 T = normalize(pin.tangent.xyz);
    float sigma = pin.tangent.w;

    T = normalize(T - N * dot(N, T));
    float3 B = normalize(cross(N, T) * sigma);

    float3 normalSample = normalMap.Sample(samplerState, pin.texcoord).xyz;
    normalSample = normalSample * 2.0f - 1.0f;
    N = normalize(normalSample.x * T
                + normalSample.y * B
                + normalSample.z * N);

    pout.normal = float4(N * 0.5f + 0.5f, 1.0f);

    // ƒ[ƒ‹ƒhÀ•W
    pout.position = float4(pin.worldPos, 1.0f);

    // MetallicRoughness
    float4 mr = metallicRoughnessMap.Sample(samplerState, pin.texcoord);
    pout.roughnessMetallic = float4(
        max(mr.g, 0.01f), // R: Roughness
        mr.b, // G: Metallic
        0.0f,
        1.0f
    );

    // Emissive
    float3 emissive = emissiveMap.Sample(samplerState, pin.texcoord).rgb;
    emissive = pow(max(emissive, 0.0001f), 2.2f);
    pout.emissive = float4(emissive, 1.0f);
    pout.occlusion = occlusionMap.Sample(samplerState, pin.texcoord).r;

    return pout;
}

{

    // –@ü³‹K‰»

        baseColor.rgb * ambientColor.rgb;

#define MAX_POINT_LIGHTS 64

cbuffer CbLights : register(b1)
{
    DirectionalLight directionalLight;
    PointLight pointLights[MAX_POINT_LIGHTS];
    int pointLightCount;
    float3 cbPadding;
};

// Gƒoƒbƒtƒ@
Texture2D gbAlbedo            : register(t0);
Texture2D gbNormal            : register(t1);
Texture2D gbPosition          : register(t2);
Texture2D gbRoughnessMetallic : register(t3);
Texture2D gbEmissive          : register(t4);
Texture2D gbOcclusion         : register(t5);

// IBL—pƒeƒNƒXƒ`ƒƒ
TextureCube diffuseIem    : register(t6);
TextureCube specularPmrem : register(t7);
Texture2D lutGgx          : register(t8);

SamplerState samplerState : register(s0);

struct VS_OUT
{
    float4 position : SV_POSITION;
    float2 texcoord : TEXCOORD;
};

float4 main(VS_OUT pin) : SV_TARGET
{
    float2 uv = pin.texcoord;

    // Gƒoƒbƒtƒ@‚©‚çî•ñ‚ğ“Ç‚İo‚·
    float4 albedo = gbAlbedo.Sample(samplerState, uv);
    float3 N = gbNormal.Sample(samplerState, uv).xyz;
    float3 worldPos = gbPosition.Sample(samplerState, uv).xyz;
    float4 roughnessMetallic = gbRoughnessMetallic.Sample(samplerState, uv);
    float3 emissive = gbEmissive.Sample(samplerState, uv).rgb;
    float  occlusion = gbOcclusion.Sample(samplerState, uv).r;

    // –@üƒfƒR[ƒh
    N = normalize(N * 2.0f - 1.0f);

    float roughness = roughnessMetallic.r;
    float metalness = roughnessMetallic.g;

    // PBRƒpƒ‰ƒ[ƒ^[ŒvZ
    float3 diffuse_reflectance = lerp(albedo.rgb, 0.0f, metalness);
    float3 F0 = lerp(0.04f, albedo.rgb, metalness);

    // ƒxƒNƒgƒ‹ŒvZ
    float3 V = normalize(cameraPosition.xyz - worldPos);
    float3 diffuse = 0;
    float3 specular = 0;
    
    // •½sŒõŒ¹
    {
        float3 L = normalize(-directionalLight.direction.xyz);
        float3 LC = directionalLight.color.rgb * directionalLight.color.a;
        float3 d = 0;
        float3 s = 0;
        DirectBRDF(diffuse_reflectance, F0, N, V, L, LC, roughness, d, s);
        diffuse += d;
        specular += s;
    }

    // ƒ|ƒCƒ“ƒgƒ‰ƒCƒg
    for (int i = 0; i < pointLightCount; i++)
    {
        float3 toLight = pointLights[i].position.xyz - worldPos;
        float dist = length(toLight);
        float range = pointLights[i].position.w;

        if (dist > range)
            continue;

        float attenuation = 1.0f - saturate(dist / range);
        attenuation *= attenuation;

        float3 L = normalize(toLight);
        float3 LC = pointLights[i].color.rgb
              * pointLights[i].color.a
              * attenuation;

        float3 d = 0;
        float3 s = 0;
        DirectBRDF(diffuse_reflectance, F0, N, V, L, LC, roughness, d, s);
        diffuse += d;
        specular += s;
    }
    
    //float3 L = normalize(-lightDirection.xyz);
    //float3 LC = lightColor.rgb * lightColor.a;

    //// ’¼ÚŒõ‚ÌPBRƒ‰ƒCƒeƒBƒ“ƒO
    //float3 diffuse = 0;
    //float3 specular = 0;
    //DirectBRDF(diffuse_reflectance, F0, N, V, L, LC,
    //           roughness, diffuse, specular);

    // ŠÔÚŒõiIBLj
    float3 iblDiffuse = IBL_Diffuse(
        diffuseIem, samplerState, N,
        diffuse_reflectance, F0, V);

    float3 iblSpecular = IBL_Specular(
        specularPmrem, lutGgx, samplerState,
        N, V, F0, roughness);

    float3 ambient = iblDiffuse + iblSpecular;

    // Occlusion“K—p
    diffuse *= occlusion;
    specular *= occlusion;
    ambient *= occlusion;

    // ÅIƒJƒ‰[‡¬
    float3 color = diffuse + specular + ambient + emissive;

    // ƒKƒ“ƒ}•â³
    color = pow(max(color, 0.0001f), 1.0f / 2.2f);

    return float4(color, albedo.a);
}

    float3 diffuse = 0;
    float3 specular = 0;

    // •½sŒõŒ¹
    {
        float3 L = normalize(-directionalLight.direction.xyz);
        float3 LC = directionalLight.color.rgb * directionalLight.color.a;
        float3 d = 0;
        float3 s = 0;
        DirectBRDF(diffuse_reflectance, F0, N, V, L, LC, roughness, d, s);
        diffuse += d;
        specular += s;
    }

    // ƒ|ƒCƒ“ƒgƒ‰ƒCƒg
    for (int i = 0; i < pointLightCount; i++)
    {
        float3 toLight = pointLights[i].position.xyz - pin.worldPos;
        float dist = length(toLight);
        float range = pointLights[i].position.w;

        if (dist > range)
            continue;

        float attenuation = 1.0f - saturate(dist / range);
        attenuation *= attenuation;

        float3 L = normalize(toLight);
        float3 LC = pointLights[i].color.rgb
              * pointLights[i].color.a
              * attenuation;

        float3 d = 0;
        float3 s = 0;
        DirectBRDF(diffuse_reflectance, F0, N, V, L, LC, roughness, d, s);
        diffuse += d;
        specular += s;
    }

    // ŠÔÚŒõ
    float3 iblDiffuse = IBL_Diffuse(
        diffuseIem, samplerState, N,
        diffuse_reflectance, F0, V);

    float3 iblSpecular = IBL_Specular(
        specularPmrem, lutGgx, samplerState,
        N, V, F0, roughness);

    float3 ambient = iblDiffuse + iblSpecular;

    // Occlusion“K—p
    diffuse *= occlusion;
    specular *= occlusion;
    ambient *= occlusion;

    // ÅIƒJƒ‰[‡¬
    float3 color = diffuse + specular + ambient + emissive;
    
    // ƒKƒ“ƒ}•â³
    color = pow(max(color, 0.0001f), 1.0f / GAMMA);

    return float4(color, baseColor.a);
}

// Diffuse IBL
// irradianceMap: –‘Oô‚İ‚İÏ‚İ‚ÌDiffuseŠÂ‹«ƒ}ƒbƒv
// N            : –@üƒxƒNƒgƒ‹iƒ[ƒ‹ƒh‹óŠÔj
float3 IBL_Diffuse(
    TextureCube irradianceMap,
    SamplerState samplerState,
    float3 N,
    float3 diffuse_reflectance,
    float3 F0,
    float3 V)
{
    // ƒtƒŒƒlƒ‹€
    float NdotV = max(0.0001f, dot(N, V));
    float3 F = F0 + (max(1.0f - 0.0f, F0) - F0) * pow(clamp(1.0f - NdotV, 0.0f, 1.0f), 5.0f);

    // kd
    float3 kd = (1.0f - F);

    // –@ü•ûŒü‚ÅIrradianceƒ}ƒbƒv‚ğƒTƒ“ƒvƒŠƒ“ƒO
    float3 irradiance = irradianceMap.Sample(samplerState, N).rgb;

    return kd * diffuse_reflectance * irradiance;
}

// Specular IBL
// prefilteredMap: roughness‚²‚Æ‚ÉMip‚ª‚ ‚é–‘OƒtƒBƒ‹ƒ^ƒŠƒ“ƒOÏ‚İƒ}ƒbƒv
// brdfLut       : GGX BRDF‚ÌLookUpTable
// N             : –@üƒxƒNƒgƒ‹
// V             : ‹üƒxƒNƒgƒ‹
// F0            : ‚’¼“üË‚ÌƒtƒŒƒlƒ‹”½Ë—¦
// roughness     : ‘e‚³
float3 IBL_Specular(
    TextureCube prefilteredMap,
    Texture2D brdfLut,
    SamplerState samplerState,
    float3 N,
    float3 V,
    float3 F0,
    float roughness)
{
    float NdotV = max(0.0001f, dot(N, V));

    // ”½Ë•ûŒü
    float3 R = reflect(-V, N);

    // roughness‚É‰‚¶‚ÄMipƒŒƒxƒ‹‚ğ‘I‘ğ
    float mipLevel = roughness * 10.0f;

    // –‘OƒtƒBƒ‹ƒ^ƒŠƒ“ƒOÏ‚İŠÂ‹«ƒ}ƒbƒv‚ğƒTƒ“ƒvƒŠƒ“ƒO
    float3 prefilteredColor = prefilteredMap.SampleLevel(
        samplerState, R, mipLevel).rgb;

    // BRDF LUT‚ğƒTƒ“ƒvƒŠƒ“ƒO
    float2 brdf = brdfLut.Sample(samplerState,
        float2(NdotV, roughness)).rg;

    // ƒtƒŒƒlƒ‹€
    float3 F = F0 + (max(1.0f - roughness, F0) - F0) *
               pow(clamp(1.0f - NdotV, 0.0f, 1.0f), 5.0f);

    return prefilteredColor * (F * brdf.x + brdf.y);
}

#ifndef PI
#define PI 3.14159265358979323846f
#endif

#ifndef GAMMA
#define GAMMA 2.2f
#endif

// ƒtƒŒƒlƒ‹€ (Schlick‚Ì‹ß—®)

// F0    : ‚’¼“üË‚Ì”½Ë—¦
// VdotH : ‹üƒxƒNƒgƒ‹‚Æƒn[ƒtƒxƒNƒgƒ‹‚Ì“àÏ
float3 CalcFresnel(float3 F0, float VdotH)
{
    return F0 + (1.0f - F0) * pow(clamp(1.0f - VdotH, 0.0f, 1.0f), 5.0f);
}

// ŠgU”½ËBRDF (³‹K‰»ƒ‰ƒ“ƒo[ƒg)

// VdotH              : ‹ü‚Æƒn[ƒtƒxƒNƒgƒ‹‚Ì“àÏ
// F0                 : ‚’¼“üË‚ÌƒtƒŒƒlƒ‹”½Ë—¦
// diffuse_reflectance: ŠgU”½Ë—¦
float3 DiffuseBRDF(float VdotH, float3 F0, float3 diffuse_reflectance)
{
    return (1.0f - CalcFresnel(F0, VdotH)) * (diffuse_reflectance / PI);
}

// D€: –@ü•ª•zŠÖ” (GGX)

// NdotH    : –@ü‚Æƒn[ƒtƒxƒNƒgƒ‹‚Ì“àÏ
// roughness: ‘e‚³
float CalcNDF(float NdotH, float roughness)
{
    float a = roughness * roughness;
    float a2 = a * a;
    float d = (NdotH * NdotH) * (a2 - 1.0f) + 1.0f;
    return a2 / (PI * d * d);
}

// G€: Šô‰½Œ¸Š€ (Smith-Schlick)

// NdotL    : –@ü‚ÆŒõŒ¹ƒxƒNƒgƒ‹‚Ì“àÏ
// NdotV    : –@ü‚Æ‹üƒxƒNƒgƒ‹‚Ì“àÏ
// roughness: ‘e‚³
float CalcGeometry(float NdotL, float NdotV, float roughness)
{
    float r = roughness * 0.5f;
    float shadowing = NdotL / (NdotL * (1.0f - r) + r);
    float masking = NdotV / (NdotV * (1.0f - r) + r);
    return shadowing * masking;
}

// ‹¾–Ê”½ËBRDF (Cook-Torranceƒ‚ƒfƒ‹)

// NdotV    : –@ü‚Æ‹üƒxƒNƒgƒ‹‚Ì“àÏ
// NdotL    : –@ü‚ÆŒõŒ¹ƒxƒNƒgƒ‹‚Ì“àÏ
// NdotH    : –@ü‚Æƒn[ƒtƒxƒNƒgƒ‹‚Ì“àÏ
// VdotH    : ‹ü‚Æƒn[ƒtƒxƒNƒgƒ‹‚Ì“àÏ
// F0       : ‚’¼“üË‚ÌƒtƒŒƒlƒ‹”½Ë—¦
// roughness: ‘e‚³
float3 SpecularBRDF(float NdotV, float NdotL, float NdotH, float VdotH,
                    float3 F0, float roughness)
{
    float D = CalcNDF(NdotH, roughness);
    float G = CalcGeometry(NdotL, NdotV, roughness);
    float3 F = CalcFresnel(F0, VdotH);
    return (D * G * F) / (NdotL * NdotV * 4.0f);
}

// ’¼ÚŒõ‚ÌPBRƒ‰ƒCƒeƒBƒ“ƒO

// diffuse_reflectance: ŠgU”½Ë—¦
// F0                 : ‚’¼“üË‚ÌƒtƒŒƒlƒ‹”½Ë—¦
// N                  : –@ü (³‹K‰»Ï‚İ)
// V                  : ‹üƒxƒNƒgƒ‹ (ƒJƒƒ‰•ûŒü, ³‹K‰»Ï‚İ)
// L                  : ŒõŒ¹•ûŒüƒxƒNƒgƒ‹ (ŒõŒ¹‚ÖŒü‚©‚¤, ³‹K‰»Ï‚İ)
// light_color        : ƒ‰ƒCƒgƒJƒ‰[
// roughness          : ‘e‚³
void DirectBRDF(
    float3 diffuse_reflectance,
    float3 F0,
    float3 N, float3 V, float3 L,
    float3 light_color,
    float roughness,
    out float3 out_diffuse,
    out float3 out_specular)
{
    float3 H = normalize(L + V);
    float NdotV = max(0.0001f, dot(N, V));
    float NdotL = max(0.0001f, dot(N, L));
    float NdotH = max(0.0001f, dot(N, H));
    float VdotH = max(0.0001f, dot(V, H));

    float3 irradiance = light_color * NdotL;

    out_diffuse = DiffuseBRDF(VdotH, F0, diffuse_reflectance) * irradiance;
    out_specular = SpecularBRDF(NdotV, NdotL, NdotH, VdotH, F0, roughness) * irradiance;
}
